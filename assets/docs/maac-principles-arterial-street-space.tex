\noindent
The Multimodal Accessibility Advisory Council proposes the following priorities
(in order) for arterial street space in the District of Columbia, aligned with our
vision for quality of life, accessibility, efficiency, and sustainability.

\vspace{4mm}

% ================= PRIORITY LADDER =================
{\color{maacnavy}\bfseries\large Priorities for arterial street space, in order\par}
\vspace{1mm}
{\color{rulegrey}\hrule height 0.8pt}
\vspace{3mm}

\prio{ADA-compliant sidewalks on both sides of the street}{, free from obstructions
(e.g.\ utility poles, tree roots, snow, trash bins)\flag{first priority}}

\prio{24/7 dedicated, protected lanes for public transit and emergency vehicles}{}

\prio{ADA pickup/dropoff zones}{ exclusively for use by disabled users}

\prio{Protected bike/scooter/power wheelchair lanes}{ separate from cars and
pedestrians (especially when sidewalks are inaccessible)}

\prio{Bike/scooter parking}{ (daylighting infrastructure) to keep bikes off sidewalks}

\prio{General motor vehicle travel lanes}{}

\prio{General (non-ADA) pickup/dropoff zones}{}

\prio{ADA-only parking}{ with 4-hour time limits}

\prio{ADA-accessible parklets and streeteries}{}

\prio{Tree boxes and green space}{}

\prio{Truly market-priced car parking}{\flag{last priority}}

\vspace{3mm}

% ================= KEY PRINCIPLES =================
{\color{maacnavy}\bfseries\large Key principles for implementation\par}
\vspace{1mm}
{\color{rulegrey}\hrule height 0.8pt}
\vspace{3mm}

\begin{tcolorbox}[
  colback=panelbg, colframe=maacnavy, boxrule=0pt,
  leftrule=3pt, arc=0pt, outer arc=0pt,
  left=6mm, right=4mm, top=2mm, bottom=2mm]
\begin{itemize}[leftmargin=5mm, itemsep=1.5pt, topsep=0pt, parsep=0pt, label={\color{maacred}\textbf{--}}]
  \item Separate vulnerable users from motor vehicles
  \item Minimize conflicts at intersections and driveways
  \item Maximize visibility for all users
  \item Be open to major changes in street design
  \item Prioritize sidewalk accessibility and safety
  \item Minimize exceptions for individual circumstances
  \item Eliminate car delay as a primary consideration
\end{itemize}
